736 // Comments about the algorithm below.
737 //
738 // Credit: This is based on an algorithm by taocpp here:
739 // https://github.com/taocpp/sequences/blob/main/include/tao/seq/make_integer_sequence.hpp
740 // but significantly simplified.
741 //
742 // We build up a sequence S by repeatedly doubling its length and sometimes adding 1 to the end.
743 // E.g. if the current S is 0...3, then we either go to 0...7 or 0...8 on the next pass.
744 // The goal is to end with S = 0...N-1.
745 // The key insight is that the times we need to add an additional digit to S correspond
746 // exactly to the 1's in the binary representation of the number N.
747 //
748 // Invariants:
749 // - digit is a power of 2
750 // - N_digit_is_1 is whether N's binary representation has a 1 in that digit's position.
751 // - end <= N
752 // - S is 0...end-1.
753 // - if digit > 0, end * digit * 2 <= N < (end+1) * digit * 2
754 //
755 // The process starts with digit > N, end = 0, and S is empty.
756 // The process concludes with digit=0, in which case, end == N and S is 0...N-1.
757
758 template <size_t digit, bool N_digit_is_1, size_t N, size_t end, size_t... S> // N_digit_is_1=false
759 struct make_index_sequence_impl
760 : make_index_sequence_impl<digit / 2, (N & (digit / 2)) != 0, N, 2 * end, S..., (S + end)...> {
761 };
762 template <size_t digit, size_t N, size_t end, size_t... S>
763 struct make_index_sequence_impl<digit, true, N, end, S...>
764 : make_index_sequence_impl<digit / 2,
765 (N & (digit / 2)) != 0,
766 N,
767 2 * end + 1,
768 S...,
769 (S + end)...,
770 2 * end> {};
771 template <size_t N, size_t end, size_t... S>
772 struct make_index_sequence_impl<0, false, N, end, S...> {
773 using type = index_sequence<S...>;
774 };
